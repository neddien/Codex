// What about Asset loaders? Loaders are cool because you can abstract loading logic using polymorphism which is convinient.

template <typename T>
concept AssetLoader = requires(T& obj, fs::FileHandle fh) {
    typename T::asset_type;
    { obj.load(fh) } -> std::is_convertible<mem::Shared<typename T::asset_type>>;
};

template <typename T>
concept ParameterizedLoader =
    AssetLoader<T> &&
    requires {
        typename T::load_params;
        requires !std::same_as<typename T::load_params, void>;
    } &&
    requires(T& obj, fs::FileHandle fh, typename T::load_params& params) {
        { obj.load(fh, params) } -> std::is_same_as<mem::Shared<typename T::asset_type>>;
    };

template <typename T, typename U = void>
struct IAssetLoader
{
    using asset_type = T;
    using load_params = U;
};

class TextureLoader : public IAssetLoader<Texture2D, Texture2D::ImportDescription>
{
    mem::Shared<Texture2D> load() {
        // load
        return {};
    }
};

template <LoadableAsset T, typename U>
mem::Shared<T> AssetManager::load(fs::FileHandle handle, const U& desc)
{
    return loaders_.get(std::type_index(typeid(T::asset_type))).load(handle, desc);
}

template <AssetLoader T>
void AssetManager::register_loader()
{
     T loader;
     loader.init();
     loaders_.[std::type_index(typeid(T::asset_type))] = std::move(loader);
}

AssetManager::register_loader<TextureLoader>();
